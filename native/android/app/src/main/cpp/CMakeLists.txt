cmake_minimum_required(VERSION 3.22.1)
project(krnode)
set(LIBNODE ${CMAKE_SOURCE_DIR}/../../../libnode)
add_library(krnode SHARED native-lib.cpp)
target_include_directories(krnode PRIVATE ${LIBNODE}/include/node)
add_library(libnode SHARED IMPORTED)
set_target_properties(libnode PROPERTIES IMPORTED_LOCATION ${LIBNODE}/bin/${ANDROID_ABI}/libnode.so)
find_library(log-lib log)
target_link_libraries(krnode libnode ${log-lib})
